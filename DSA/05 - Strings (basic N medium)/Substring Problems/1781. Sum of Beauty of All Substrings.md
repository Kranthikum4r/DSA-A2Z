```
class Solution {
public:
    int beautySum(string s) {
        int n = s.length();
        int total = 0;

        for(int i = 0; i < n; i++) {
            vector<int> freq(26, 0);

            for(int j = i; j < n; j++) {
                freq[s[j] - 'a']++;

                int m = INT_MAX, M = 0;
                for(int k = 0; k < 26; k++) {
                    if(freq[k] > 0) {
                        m = min(m, freq[k]);
                        M = max(M, freq[k]);
                    }
                }

                total += (M - m);
            }
        }
        return total;
    }
};
```
