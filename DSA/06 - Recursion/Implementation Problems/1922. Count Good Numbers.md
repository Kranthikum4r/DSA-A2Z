class Solution {
public:
    const long long MOD = 1e9 + 7;

    long long modpower(long long base, long long exponent) {
        long long result = 1;
        base %= MOD;

        while(exponent > 0) {
            // If the current bit of the exponent is 1,
            // include the current base in the answer.
            if(exponent % 2 == 1) {
                result = (result * base) % MOD;
            }

            base = (base * base) % MOD;

            //move to next bit
            exponent /= 2;
        }
        return result;
    }
    int countGoodNumbers(long long n) {
        long long evenPositions = (n + 1) / 2;

        long long oddPositions = n / 2;

        // even positions(even numbers) -> 0, 2, 4, 6, 8 (5 choices)
        // odd positions(prime numbers) -> 2, 3, 5, 7 (4 choices)
        long long evenways = modpower(5, evenPositions);
        long long oddways = modpower(4, oddPositions);

        return (evenways * oddways) % MOD;
    }
};
