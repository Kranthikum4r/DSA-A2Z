```
class Solution {
private:
    void generate(string cur, int open, int close, int n, vector<string>& ans) {
        if(cur.length() == 2 * n) {
            ans.push_back(cur);
            return;
        }

        if(open < n) {
            generate(cur + '(', open + 1, close, n, ans);
        }
        if(close < open) {
            generate(cur + ')', open, close + 1, n, ans);
        }
    }
public:
    vector<string> generateParenthesis(int n) {
        vector<string> ans;

        generate("", 0, 0, n , ans);

        return ans;
    }
};
```
